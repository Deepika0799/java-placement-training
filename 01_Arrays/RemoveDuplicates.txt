import java.util.Arrays;

public class RemoveDuplicates {
    public static void main(String[] args) {

        int[] a = {1, 1, 2, 2, 3, 4, 4};

        int j = 0;

        for (int i = 1; i < a.length; i++) {

            if (a[i] != a[j]) {
                j++;
                a[j] = a[i];
            }
        }

        System.out.println(Arrays.toString(Arrays.copyOf(a, j + 1)));
    }
}
